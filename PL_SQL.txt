mysql> CREATE FUNCTION CalculateFine(roll_no INT, doi DATE)
    -> RETURNS DECIMAL(10,2)
    -> DETERMINISTIC
    -> BEGIN
    ->     DECLARE noofdays INT;
    ->     DECLARE fine_amt DECIMAL(10,2);
    ->     DECLARE gtfifty INT;
    -> 
    ->     -- Calculate days difference
    ->     SET noofdays = DATEDIFF(CURDATE(), doi);
    -> 
    ->     -- Fine calculation logic
    ->     IF noofdays < 15 THEN
    ->         SET fine_amt = 0;
    ->     ELSEIF noofdays BETWEEN 15 AND 30 THEN
    ->         SET fine_amt = noofdays * 5;
    ->     ELSE
    ->         SET gtfifty = noofdays - 30;
    ->         SET fine_amt = (30 * 5) + (gtfifty * 50);
    ->     END IF;
    -> 
    ->     RETURN fine_amt;
    -> END$$

mysql> CREATE PROCEDURE ReturnBook(IN p_roll_no INT, IN p_return_date DATE)
    -> BEGIN
    ->     DECLARE doi DATE;
    ->     DECLARE fine_amt DECIMAL(10,2);
    -> 
    ->     -- Get date of issue
    ->     SELECT DateOfIssue INTO doi
    ->     FROM Borrower
    ->     WHERE Rollin = p_roll_no;
    -> 
    ->     -- Calculate fine
    ->     SET fine_amt = CalculateFine(p_roll_no, doi);
    -> 
    ->     -- Update status to Returned
    ->     UPDATE Borrower
    ->     SET Status = 'R'
    ->     WHERE Rollin = p_roll_no;
    -> 
    ->     -- If fine exists, insert record into Fine table
    ->     IF fine_amt > 0 THEN
    ->         INSERT INTO Fine (Roll_no, DateOfReturn, Amt)
    ->         VALUES (p_roll_no, p_return_date, fine_amt);
    ->     END IF;
    -> END$$
