mysql> DELIMITER $$
mysql> CREATE PROCEDURE GenerateCertificates()
    -> BEGIN
    ->     DECLARE v_student_id INT;
    ->     DECLARE v_event_id INT;
    ->     DECLARE done INT DEFAULT 0;
    -> 
    ->     DECLARE cur CURSOR FOR
    ->         SELECT r.student_id, r.event_id
    ->         FROM Registration r
    ->         JOIN Event e ON r.event_id = e.event_id
    ->         WHERE e.end_date < CURDATE();
    -> 
    ->     DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;
    -> 
    ->     OPEN cur;
    ->     read_loop: LOOP
    ->         FETCH cur INTO v_student_id, v_event_id;
    ->         IF done = 1 THEN
    ->             LEAVE read_loop;
    ->         END IF;
    -> 
    ->         IF NOT EXISTS (
    ->             SELECT 1 FROM Certificate
    ->             WHERE student_id = v_student_id AND event_id = v_event_id
    ->         ) THEN
    ->             INSERT INTO Certificate(student_id, event_id, issue_date)
    ->             VALUES(v_student_id, v_event_id, CURDATE());
    ->         END IF;
    ->     END LOOP;
    ->     CLOSE cur;
    -> END$$
